nginx ​

位置 /usr/local/nginx/ 现在用的

nginx
worker_processes  1;

error_log  /var/log/nginx/error.log warn;
pid        /var/run/nginx.pid;

events {
    # 可以根据业务并发量适当调高
    worker_connections  1024;
}

http {
    include       mime.types;
    default_type  application/octet-stream;

    # 高效传输文件
    sendfile on;

    # 长连接超时时间
    keepalive_timeout 65;

    # 单连接最大请求数，提高长连接复用率
    keepalive_requests 100000;

    # 限制body大小
    client_max_body_size 100m;
    client_header_buffer_size 32k;
    client_body_buffer_size   512k;

    # 开启静态资源压缩（最小改动：替换 gzip_static）
    gzip on;

    # 连接数限制 (防御类配置)
    limit_conn_zone $binary_remote_addr zone=perip:10m;
    limit_conn_zone $server_name zone=perserver:10m;

    # 隐藏 nginx 版本号
    server_tokens off;

    log_format main
        '$remote_addr - $remote_user [$time_local] "$request" '
        '$status $body_bytes_sent "$http_referer" '
        '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    include /usr/local/nginx/default.d/*.conf;
}

ruoyiweb.conf ​

nginx
# =================== 上游服务 ===================
upstream server {
    ip_hash;
    # 这里改为 Docker 容器映射端口
    server 127.0.0.1:40001;
}

upstream monitor-admin {
    server 127.0.0.1:9090;
}

upstream snailjob-server {
    server 127.0.0.1:8800;
}

# =================== 站点配置 ===================
server {
    listen 80;
    server_name localhost;

    # 禁止外网访问 actuator
    location ~ ^(/[^/]*)?/actuator.*(/.*)?$ {
        return 403;
    }

    # 根路径前端静态资源
    location / {
        root /root/ruoyiweb;
        index index.html index.htm;
        try_files $uri $uri/ /index.html;
    }

    # 生产 API 代理
    location /prod-api/ {
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header REMOTE-HOST $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_read_timeout 86400s;

        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        proxy_buffering off;
        proxy_cache off;

        limit_conn perip 20;
        limit_conn perserver 500;

        proxy_pass http://server/;
    }

    # admin 代理
    location /admin/ {
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header REMOTE-HOST $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_buffering off;
        proxy_cache off;

        proxy_pass http://monitor-admin/admin/;
    }

    # snail-job 代理
    location /snail-job/ {
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header REMOTE-HOST $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        proxy_buffering off;
        proxy_cache off;

        proxy_pass http://snailjob-server/snail-job/;
    }

    # 错误页面
    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /root/ruoyiweb;
    }
}

stcode-docs-8881.conf ​

nginx
server {
    listen       8881;
    server_name  localhost;
    charset utf-8;

    # ===== CORS（统一放在 server 级）=====
    add_header Access-Control-Allow-Origin "*" always;
    add_header Access-Control-Allow-Credentials "true" always;
    add_header Access-Control-Allow-Methods "GET, POST, OPTIONS" always;
    add_header Access-Control-Allow-Headers
        "DNT,X-CustomHeader,Keep-Alive,User-Agent,X-Requested-With,If-Modified-Since,Cache-Control,Content-Type,Authorization"
        always;

    # ===== OPTIONS 预检（这里只 return）=====
    if ($request_method = OPTIONS) {
        return 204;
    }

    # ===== /stcode-docs =====
    location /stcode-docs/ {
        alias /root/stcode-docs/;
        index index.html index.htm;
        autoindex on;
        autoindex_exact_size on;
        autoindex_localtime on;
    }

    # ===== /stcode-docs-new =====
    location /stcode-docs-new/ {
        alias /root/stcode-docs-new/;
        index index.html index.htm;
        autoindex on;
        autoindex_exact_size on;
        autoindex_localtime on;
    }
}
